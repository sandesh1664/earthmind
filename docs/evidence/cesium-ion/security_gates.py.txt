"""Check credential containment without printing configured values."""
from __future__ import annotations
import json
from pathlib import Path
import stat
import subprocess
import sys
from dotenv import dotenv_values

root=Path.cwd()
settings=dotenv_values(root/'.env')
values={value.encode() for name,value in settings.items() if value and len(value)>=8 and any(word in name for word in ('TOKEN','PASSWORD','SECRET'))}
assert values,'No configured secrets available for containment check'
items=subprocess.check_output(['git','ls-files','--cached','--others','--exclude-standard','-z']).decode().split('\0')
scanned=0
matches=[]
for name in set(items):
    path=root/name
    if not name or not path.is_file():continue
    payload=path.read_bytes();scanned+=1
    if any(value in payload for value in values):matches.append(name)
assert not matches, f'Configured credential in candidate file(s): {matches}'
assert subprocess.call(['git','check-ignore','--quiet','.env'])==0
assert stat.S_IMODE((root/'.env').stat().st_mode)==0o600
static_files=list((root/'apps/web/.next/static').rglob('*'))
static_count=0
for path in static_files:
    if path.is_file():
        static_count+=1
        assert not any(value in path.read_bytes() for value in values), 'Configured credential in built static asset'
evidence=Path(__file__).resolve().parent
evidence_count=0
for path in evidence.rglob('*'):
    if path.is_file():
        evidence_count+=1
        assert not any(value in path.read_bytes() for value in values), 'Configured credential in verification evidence'
if '--staged' in sys.argv:
    staged=subprocess.check_output(['git','diff','--cached','--name-only','-z']).decode().split('\0')
    for name in staged:
        if not name:continue
        result=subprocess.run(['git','show',':'+name],capture_output=True,check=True)
        assert not any(value in result.stdout for value in values), 'Configured credential in staged blob'
print(json.dumps({'configuredValuesChecked':len(values),'candidateFiles':scanned,'hostStaticAssets':static_count,'evidenceFiles':evidence_count,'credentialMatches':0,'ignoredEnv':True,'envMode':'0600','stagedChecked':'--staged' in sys.argv}))
